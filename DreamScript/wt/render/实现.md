```jsx@script
// 剧本件 —— **纯渲染（只读）**：认领整篇（h1 节），题头页 + Letter 纸面 + 分页 + 排版。
// 契约 { node, read }；不挂任何交互、不写回 md（就地编辑能力已移除）。
const FIELD = { 内外: 1, 地点: 1, 时间: 1 };
const META = { 编剧: 1, 稿次: 1, 日期: 1 };
const INDOOR = { 内: "INT.", 外: "EXT.", 内外: "INT./EXT." };
const IO_R = { "INT.": "内", "EXT.": "外", "INT./EXT.": "内外", "EXT./INT.": "内外", "内景": "内", "外景": "外", "内": "内", "外": "外" };
const CLS = { 动作: "sc-action", 段落: "sc-action", 镜头: "sc-shot", 转场: "sc-trans", 角色: "sc-char", 副语言: "sc-paren", 台词: "sc-dialog" };
const CPL = { 动作: 61, 段落: 61, 镜头: 61, 转场: 61, 角色: 61, 台词: 35, 副语言: 31 };
const PRE = { 转场: 1, 角色: 1 };
const POST = { 动作: 1, 段落: 1, 镜头: 1, 转场: 1, 台词: 1 };
const CAP = 54;

const val = (read, n) => String(read.contentValueOf(n) ?? "").trim();
const slugOf = (f) => [INDOOR[f["内外"]] || f["内外"] || "", f["地点"] || "", f["时间"] ? "- " + f["时间"] : ""].filter(Boolean).join(" ");

/** 整篇 → { meta, blocks }：块按树序，元素带同 tag 序号。 */
export function layout(read, root) {
  const meta = {}; const blocks = [];
  for (const k of read.kidsOf([root])) {
    const t = read.textOf(k) || "";
    if (META[t]) { meta[t] = val(read, k); continue; }
    if (read.isSectionShell(k.nt)) {
      const fields = {}; const els = []; const seen = {};
      for (const c of read.kidsOf([k])) {
        const ct = read.textOf(c) || "";
        if (FIELD[ct]) { fields[ct] = val(read, c); continue; }
        if (!ct) continue;
        const idx = seen[ct] || 0; seen[ct] = idx + 1;
        els.push({ tag: ct, text: val(read, c), idx: idx });
      }
      blocks.push({ scene: read.textOf(k) || "", slug: slugOf(fields), els: els });
      continue;
    }
    if (t) blocks.push({ scene: null, slug: "", els: [{ tag: t, text: val(read, k), idx: 0 }] });
  }
  return { meta, blocks };
}

/** 块 → 排版行（带行数估算）。 */
export function rowsOf(blocks) {
  const rows = [];
  for (const b of blocks) {
    if (b.slug) rows.push({ kind: "slug", scene: b.scene, tag: "", idx: 0, text: b.slug.toUpperCase(), n: 1, pre: 1, post: 1 });
    for (const e of b.els) {
      const cpl = CPL[e.tag] || 61;
      rows.push({ kind: "el", scene: b.scene, tag: e.tag, idx: e.idx, text: e.text, n: Math.max(1, Math.ceil(e.text.length / cpl)), pre: PRE[e.tag] || 0, post: POST[e.tag] || 0 });
    }
  }
  return rows;
}

/** 分页：按行预算切；场景头不与上文分离。 */
export function paginate(rows, cap) {
  const pages = []; let cur = []; let used = 0;
  const flush = () => { if (cur.length) pages.push(cur); cur = []; used = 0; };
  for (const r of rows) {
    const need = (cur.length ? r.pre : 0) + r.n + r.post;
    if (cur.length && used + need > cap) flush();
    if (cur.length) used += r.pre;
    cur.push(r); used += r.n + r.post;
  }
  flush();
  for (let i = 0; i < pages.length - 1; i += 1) {
    while (pages[i].length > 1 && pages[i][pages[i].length - 1].kind === "slug") pages[i + 1].unshift(pages[i].pop());
  }
  return pages;
}

/** 一行 = 一个静态 div（不挂事件、不接受输入）。 */
const rowEl = (r, cls, key) => (<div key={key} className={cls} data-raw={r.text}>{r.text}</div>);

export default function Script({ node, read }) {
  const docName = read.textOf(node) || "";
  const data = layout(read, node);
  const meta = data.meta;
  const pages = paginate(rowsOf(data.blocks), CAP);
  return (
    <div className="sc-root">
      <section className="sc-sheet sc-titlepage">
        {rowEl({ kind: "title", tag: "", idx: 0, text: docName }, "sc-title", "title")}
        <div className="sc-by">Written by</div>
        {rowEl({ kind: "meta", tag: "编剧", idx: 0, text: meta["编剧"] || "" }, "sc-author", "meta|编剧")}
        <div className="sc-draft">
          {rowEl({ kind: "meta", tag: "稿次", idx: 0, text: meta["稿次"] || "" }, "sc-draft-line", "meta|稿次")}
          {rowEl({ kind: "meta", tag: "日期", idx: 0, text: meta["日期"] || "" }, "sc-draft-line", "meta|日期")}
        </div>
      </section>
      {pages.map((page, pi) => (
        <section className="sc-sheet" key={"p" + pi}>
          {pi > 0 ? <div className="sc-pageno">{pi + 1}.</div> : null}
          {page.map((r, ri) => rowEl(r, r.kind === "slug" ? "sc-slug" : (CLS[r.tag] || "sc-action"), (r.scene === null ? "doc" : r.scene) + "|" + r.tag + "|" + r.idx + "|" + ri))}
        </section>
      ))}
    </div>
  );
}
```
