const ELEM = { ACTION: "action", LYRIC: "lyric", TRANS: "transition", PAREN: "parenthetical", DIAL: "dialogue", SHOT: "shot", GEN: "gen", BLANK: "blank", OMIT: "omit", CAST: "cast", NOTE: "note", BREAK: "pagebreak", NUM: "num", IE: "ie", LOC: "loc", TIME: "time" };
const SUBS = ["CENTER", "RIGHT", "DUAL", "UPPER"];
const F3 = String.fromCharCode(96).repeat(3);   // 三个反引号：不能写字面量（会把外层 md 围栏切断）
const TITLEKEYS = ["TITLE", "CREDIT", "WRITER", "SOURCE", "DRAFT", "DATE", "CONTACT", "NOTES", "COPYRIGHT", "SUB"];
const ALIAS = { 动作: "ACTION", 转场: "TRANS", 居中: "ACTION_CENTER", 歌词: "LYRIC", 台词: "DIAL", 副语言: "PAREN", 批注: "NOTE", 分页: "BREAK", 角色: "CHAR", 镜头: "SHOT", 段落: "GEN", 空段: "BLANK", 内外: "IE", 地点: "LOC", 时间: "TIME", 编剧: "WRITER", 稿次: "DRAFT", 日期: "DATE", 联系: "CONTACT", 副题: "SUB" };

/** tag → {base, subs}；未知的 base 返回 null（由调用方决定当动作还是当角色名） */
function splitTag(raw) {
  let t = String(raw || "").trim();
  if (ALIAS[t]) t = ALIAS[t];
  const parts = t.split("_");
  let base = parts[0];
  const subs = [];
  for (const p of parts.slice(1)) {
    const u = p.toUpperCase();
    if (SUBS.indexOf(u) >= 0) subs.push(u);
    else if (ELEM[base + "_" + p]) base = base + "_" + p;
    else subs.push("?" + p);
  }
  return { tag: t, base: base, subs: subs, known: !!ELEM[base] };
}

/** md ⇒ IR（规范写法与别名都吃） */
function parseMd(text) {
  const L = clean(text).split(String.fromCharCode(10));
  const ir = { title: { name: "", fields: [] }, scenes: [] };
  let i = 0, inTitle = true, scene = null, para = [], prevLine = -1;   // prevLine：上一非空行行号（判「相邻 = 同段」）
  const pushField = (arr, key, val) => {
    const last = arr[arr.length - 1];
    if (last && last.key === key) { last.lines.push(val); return; }
    arr.push({ key: key, lines: [val] });
  };
  const flushPara = () => { if (para.length) { add({ type: "action", text: para.join(String.fromCharCode(10)), subs: [] }, true); para = []; } };
  const add = (el, asTop) => {
    if (!scene) { scene = { heading: { text: "", num: null, special: false }, fields: [], elements: [] }; ir.scenes.push(scene); }
    scene.elements.push(el);
  };
  while (i < L.length) {
    const line = L[i];
    if (!line.trim()) { flushPara(); i += 1; prevLine = -1; continue; }
    if (/^#\s+/.test(line)) { ir.title.name = line.replace(/^#\s+/, "").trim(); i += 1; continue; }
    if (/^---\s*$/.test(line)) { inTitle = false; i += 1; continue; }
    if (/^\*\*\*\s*$/.test(line)) { flushPara(); add({ type: "pagebreak", text: "", subs: [] }); i += 1; continue; }
    if (/^##\s+/.test(line)) {
      flushPara(); inTitle = false;
      let t = line.replace(/^##\s+/, "").trim(), num = null;
      const m = /#([A-Za-z0-9.\-]+)#\s*$/.exec(t); if (m) { num = m[1]; t = t.replace(/#[A-Za-z0-9.\-]+#\s*$/, "").trim(); }
      const cn = /^(\d+)\.\s*(.*)$/.exec(t); if (cn) { num = num || cn[1]; t = cn[2]; }
      scene = { heading: { text: t, num: num, special: false }, fields: [], elements: [] }; ir.scenes.push(scene); i += 1; continue;
    }
    if (/^###\s+/.test(line)) { flushPara(); add({ type: "special", text: line.replace(/^###\s+/, "").trim(), subs: [] }); i += 1; continue; }
    if (/^####\s+/.test(line)) { flushPara(); add({ type: "section", text: line.replace(/^####\s+/, "").trim(), subs: [] }); i += 1; continue; }
    if (/^>\s?/.test(line)) { flushPara(); add({ type: "transition", text: line.replace(/^>\s?/, "").trim(), subs: [] }); i += 1; continue; }
    if (/^<!--/.test(line)) {
      let buf = line, j = i;
      while (!/-->/.test(buf) && j + 1 < L.length) { j += 1; buf += " " + L[j].trim(); }
      const inner = buf.replace(/^<!--/, "").replace(/-->.*$/, "").trim();
      flushPara();
      const alt = /^ALT[:：]\s*(.*)$/.exec(inner);
      add(alt ? { type: "alt", text: alt[1], subs: [] } : { type: "note", text: inner, subs: [] });
      i = j + 1; continue;
    }
    if (line.startsWith(F3)) {
      const owner = line.startsWith(F3) ? (line.slice(3).trim() || "note") : "note";
      const body = []; i += 1;
      while (i < L.length && !L[i].startsWith(F3)) { body.push(L[i]); i += 1; }
      i += 1; flushPara();
      add({ type: "note", owner: owner, text: body.join(String.fromCharCode(10)), multiline: true, subs: [] });
      continue;
    }
    const indented = /^\s{2,}(?:[-*]\s+)?(.*)$/.exec(line);   // 标记必须跟空格，否则 *低沉* 会被吃掉星号
    if (indented && scene && scene.elements.length) {
      const cur = scene.elements[scene.elements.length - 1];
      if (cur && cur.type === "speech") {
        const body = indented[1].trim();
        const kv = /^([A-Za-z_]{1,24}|[^:：]{1,6})[:：]\s*(.*)$/.exec(body);
        const sb = kv ? splitTag(kv[1]) : null;
        if (sb && sb.base === "PAREN") { cur.lines.push({ type: "parenthetical", text: kv[2] }); i += 1; continue; }
        if (sb && sb.base === "DIAL") { cur.lines.push({ type: "dialogue", text: kv[2] }); i += 1; continue; }
        const ty = isParenLine(body) || isStarLine(body) ? 'parenthetical' : 'dialogue';
        const lastLn = cur.lines[cur.lines.length - 1];
        if (lastLn && lastLn.type === ty && prevLine === i - 1) lastLn.text = lastLn.text + String.fromCharCode(10) + body;   // 相邻行 = 同一段
        else cur.lines.push({ type: ty, text: body });
        prevLine = i; i += 1; continue;
      }
    }
    const top = /^([-*])\s+(.*)$/.exec(line);
    if (top) {
      flushPara();
      const mark = top[1], body = top[2];
      const kvT = /^([^:：]+)[:：]\s*(.*)$/.exec(body);
      const kv = kvT ? [kvT[0], kvT[1].trim(), kvT[2]] : /^([A-Za-z_]{1,24}|[^:：]{1,8})[:：]\s*(.*)$/.exec(body);
      const rawTag = kv ? String(kv[1]).trim() : "";
      const val = kv ? kv[2] : body;
      const spPre = splitTag(rawTag);
      if (inTitle && !spPre.known) { pushField(ir.title.fields, (ALIAS[rawTag] || rawTag || body).toUpperCase(), val); i += 1; continue; }
      const sp = splitTag(rawTag);
      const cueDual = /\^\s*$/.test(rawTag);
      const nameOnly = rawTag.replace(/\s*\^\s*$/, "").trim();
      const isCue = sp.base === "CHAR" || (!sp.known && nameOnly && !inTitle && mark === "-");   // 两族：- 族=说话，* 族=元素（词表外则报手滑）   // 已知 tag ⇒ 元素；否则 ⇒ 角色名
      if (isCue) {
        const nm = sp.base === "CHAR" ? val.trim() : nameOnly;
        const ext = /^(.+?)\s*([（(][^）)]*[）)])?\s*$/.exec(nm) || [];
        add({ type: "speech", cue: { name: (ext[1] || nm).trim(), ext: (ext[2] || "").trim(), dual: cueDual || sp.subs.indexOf("DUAL") >= 0 }, lines: [] });
        i += 1; continue;
      }
      if (sp.base === "NUM") { if (scene) scene.heading.num = val.trim(); i += 1; continue; }
      if (sp.base === "IE" || sp.base === "LOC" || sp.base === "TIME") { if (scene && !scene.elements.length) scene.fields.push({ key: sp.base.toUpperCase(), text: val.trim() }); i += 1; continue; }
      const ty = sp.known ? ELEM[sp.base] : "action";
      const prevEl = scene && scene.elements.length ? scene.elements[scene.elements.length - 1] : null;
      if ((ty === "parenthetical" || ty === "dialogue") && prevEl && prevEl.type === "speech") {
        prevEl.lines.push({ type: ty, text: val.trim() }); i += 1; continue;   // 旧体例的平铺写法：接回说话块
      }
      if (mark === "*" && !sp.known) { add({ type: "action", text: body.trim(), subs: [], unknownTag: rawTag }); i += 1; continue; }
      add({ type: ty, text: val.trim(), subs: sp.subs });
      i += 1; continue;
    }
    para.push(line.trim()); i += 1;
  }
  flushPara();
  return ir;
}

/** IR ⇒ md（只输出规范写法） */
function emitMd(ir) {
  const out = ["# " + (ir.title.name || "未命名剧本")];
  for (const f of ir.title.fields) for (const ln of f.lines) out.push("* " + f.key + ": " + ln);
  out.push("", "---", "");
  const body = [];
  for (const sc of ir.scenes) {
    if (body.length) body.push("");
    const fld = {}; for (const f of sc.fields) fld[f.key] = f.text;
    const IO = { "内": "INT.", "外": "EXT.", "内外": "INT./EXT.", "内景": "INT.", "外景": "EXT." };
    let slug = sc.heading.text;
    if ((fld.IE || fld.LOC || fld.TIME) && !/^(INT|EXT|EST|I\/E)/i.test(slug)) {
      slug = [IO[fld.IE] || fld.IE || "", fld.LOC || "", fld.TIME ? "- " + fld.TIME : ""].filter(Boolean).join(" ").trim();
      sc.heading.text = slug; if (fld.TIME) sc.heading.time = fld.TIME;
    }
    if (slug) body.push("## " + slug + (sc.heading.num ? " #" + sc.heading.num + "#" : ""));
    for (const el of sc.elements) {
      const gap = el.type === "speech" || el.type === "note" || el.type === "alt" ? true : true;
      if (gap && body.length && body[body.length - 1] !== "") body.push("");
      if (el.type === "action") body.push((el.subs || []).length ? "* ACTION_" + el.subs.join("_") + ": " + el.text : el.text);
      else if (el.type === "speech") {
        body.push("- " + el.cue.name + (el.cue.ext ? " " + el.cue.ext : "") + (el.cue.dual ? " ^" : "") + ":");
let prevTy = null;
        for (const ln of el.lines) {
    if (ln.type === 'dialogue' && prevTy === 'dialogue') body.push('');   // 两段台词之间必须空行分段
          const put = (s) => String(s).split(String.fromCharCode(10)).forEach((one) => body.push('    ' + one));   // 段内每行都要缩进
    const ambiguous = ln.type === 'dialogue' && (isStarLine(ln.text) || isParenLine(ln.text));
    if (ambiguous) put('- DIAL: ' + ln.text);
    else if (ln.type === 'parenthetical' && !isParenLine(ln.text) && !isStarLine(ln.text)) put('*' + ln.text + '*');
    else put(ln.text);
          prevTy = ln.type;
  }
      }
      else if (el.type === "lyric") body.push("* LYRIC" + ((el.subs || []).length ? "_" + el.subs.join("_") : "") + ": " + el.text);
      else if (el.type === "transition") body.push("> " + el.text);
      else if (el.type === "special") body.push("### " + el.text);
      else if (el.type === "section") body.push("#### " + el.text);
      else if (el.type === "pagebreak") body.push("***");
      else if (el.type === "note" || el.type === "alt") {
        if (el.multiline || String(el.text).includes(String.fromCharCode(10))) { body.push(F3 + (el.owner || "note")); for (const l2 of String(el.text).split(String.fromCharCode(10))) body.push(l2); body.push(F3); }
        else body.push(el.type === "alt" ? "<!-- ALT: " + el.text + " -->" : "<!-- " + el.text + " -->");
      }
      else if (el.type === "gen" || el.type === "blank" || el.type === "omit" || el.type === "cast" || el.type === "shot") body.push("* " + el.type.toUpperCase() + (el.text ? ": " + el.text : ":"));
      else body.push(el.text);
    }
  }
  return out.concat(body).join(String.fromCharCode(10)).replace(new RegExp(String.fromCharCode(10) + "{3,}", "g"), String.fromCharCode(10) + String.fromCharCode(10)).trimEnd() + String.fromCharCode(10);
}

/** 扁平元素序列（判据与渲染都用） */
function tokens(ir) {
  const t = [];
  for (const f of ir.title.fields) t.push("title." + f.key + "=" + f.lines.join("/"));
  for (const sc of ir.scenes) {
    if (sc.heading.text) t.push("scene:" + sc.heading.text + (sc.heading.num ? "#" + sc.heading.num : ""));
    else t.push("head:");
    for (const f of sc.fields) t.push("field." + f.key + "=" + f.text);
    for (const el of sc.elements) {
      if (el.type === "speech") {
        t.push("speech:" + el.cue.name + (el.cue.ext ? " " + el.cue.ext : "") + (el.cue.dual ? " ^" : ""));
        for (const ln of el.lines) t.push(ln.type + ":" + ln.text);
      } else t.push(el.type + ":" + (el.text || "") + ((el.subs || []).length ? " [" + el.subs.join(",") + "]" : ""));
    }
  }
  return t;
}

/** Fountain / 外部剧本文本 ⇒ IR（导入路径：宽松） */
function parseFountain(text) {
  const L = clean(text).split(String.fromCharCode(10));
  const ir = { title: { name: "", fields: [] }, scenes: [] };
  let i = 0, speech = false, scene = null, para = [];
  const flush = () => { if (para.length) { add({ type: "action", text: para.join(String.fromCharCode(10)), subs: [] }); para = []; } };
  const add = (el) => {
    if (!scene) { scene = { heading: { text: "(无场景头)", num: null, special: false }, fields: [], elements: [] }; ir.scenes.push(scene); }
    scene.elements.push(el);
  };
  while (i < L.length && /^[A-Za-z][A-Za-z ]*:/.test(L[i])) {
    const m = /^([A-Za-z][A-Za-z ]*):\s*(.*)$/.exec(L[i]);
    const cur = { key: m[1].toUpperCase(), lines: m[2] ? [m[2]] : [] };
    i += 1;
    while (i < L.length && /^\s{3,}\S/.test(L[i])) { cur.lines.push(L[i].trim()); i += 1; }
    ir.title.fields.push(cur);
  }
  if (ir.title.fields.length && ir.title.fields[0].key === "TITLE") ir.title.name = ir.title.fields[0].lines[0];
  while (i < L.length) {
    const line = L[i];
    if (!line.trim()) { flush(); speech = false; i += 1; continue; }
    if (/^\/{0,1}\*/.test(line) && /^\/\*/.test(line)) { i += 1; while (i < L.length && !/\*\/$/.test(L[i])) i += 1; i += 1; continue; }
    if (/^={3,}\s*$/.test(line)) { flush(); add({ type: "pagebreak", text: "", subs: [] }); i += 1; continue; }
    if (/^~/.test(line)) { flush(); add({ type: "lyric", text: line.replace(/^~\s*/, ""), subs: [] }); i += 1; continue; }
    if (/^>/.test(line)) {
      flush();
      const centered = /<\s*$/.test(line);
      const txt = line.replace(/^>\s*/, "").replace(/\s*<\s*$/, "");
      add(centered ? { type: "action", text: txt, subs: ["CENTER"] } : { type: "transition", text: txt, subs: [] });
      i += 1; continue;
    }
    if (/^#/.test(line)) { flush(); add({ type: "section", text: line.replace(/^#+\s*/, ""), subs: [] }); i += 1; continue; }
    if (/^\.[A-Za-z0-9]/.test(line)) { flush(); scene = { heading: { text: line.slice(1).trim(), num: null, special: true }, fields: [], elements: [] }; ir.scenes.push(scene); i += 1; continue; }
    if (/^(INT|EXT|EST|INT\.\/EXT|INT\/EXT|I\/E)[.\s]/i.test(line)) {
      flush();
      let t = line.trim(), num = null;
      const m = /#([A-Za-z0-9.\-]+)#\s*$/.exec(t); if (m) { num = m[1]; t = t.replace(/#[A-Za-z0-9.\-]+#\s*$/, "").trim(); }
      scene = { heading: { text: t, num: num, special: false }, fields: [], elements: [] }; ir.scenes.push(scene); i += 1; speech = false; continue;
    }
    if (/^[A-Z0-9 .\-()&]*TO:\s*$/.test(line)) { flush(); add({ type: "transition", text: line.trim(), subs: [] }); i += 1; speech = false; continue; }
    const forced = /^@/.test(line);
    const cand = line.replace(/^@/, "").replace(/\s*\^\s*$/, "").trim();
    const caps = /^[^a-z]*[A-Z][^a-z]*$/.test(cand);
    if ((forced || caps) && !speech && L[i + 1] && L[i + 1].trim()) {
      const m = /^(.+?)\s*([（(][^）)]*[）)])?\s*$/.exec(cand) || [];
      add({ type: "speech", cue: { name: (m[1] || cand).trim(), ext: (m[2] || "").trim(), dual: /\^\s*$/.test(line) }, lines: [] });
      speech = true; i += 1; continue;
    }
    if (speech) {
      const cur = scene.elements[scene.elements.length - 1];
      cur.lines.push({ type: /^[（(][^）)]*[）)]$/.test(line.trim()) ? "parenthetical" : "dialogue", text: line.trim() });   // Fountain 的整行 *…* 是斜体强调，不是副语言
      i += 1; continue;
    }
    para.push(line.trim()); i += 1;
  }
  flush();
  return ir;
}
/** 给 IR 每个节点补 addr（可执行选择器；多行说话块按「块地址 + 行号」约定） */
function assignAddr(ir) {
  // 地址文法实测（见 report）：锚点之后**只能有一组方括号**；裸名字当整条选择器会静默零命中
  //   ⇒ 写回用「名字锚 + 一组」："场景名"..[p][j] / "场景名"._b[b.t=="TAG"][k]
  //   ⇒ 读节体另用 ..*[h2][i]（单组，实测可用）
  const headAnchor = JSON.stringify(ir.title.name || "未命名剧本");
  let hp = 0, hb = 0, h3 = 0, h4 = 0;
  const useHead = (el) => {
    if (el.type === "speech") { el.addr = headAnchor + "._b[b.t==" + JSON.stringify(el.cue.name) + "][" + (hb++) + "]"; return; }
    if (el.type === "action") { el.addr = headAnchor + "..[p][" + (hp++) + "]"; return; }
    if (el.type === "transition") { el.addr = headAnchor + "..[bq][0]"; return; }
    el.addr = headAnchor + "._b[b.t==" + JSON.stringify(el.type.toUpperCase()) + "][0]";
  };
  let h2 = 0;   // h2 只数真正的场景头（文首「无场景头」的伪场景不占 ##）
  ir.scenes.forEach((sc, si) => {
    const anchor = JSON.stringify(sc.heading.text);
    sc.heading.addr = anchor;                       // 写回锚（名字）
    if (sc.heading.text !== "") sc.heading.readAddr = "..*[h2][" + (h2++) + "]";   // 读节体（单组方括号）
    if (sc.heading.text === "") { sc.elements.forEach((el) => { useHead(el); el.head = true; }); return; }
    const seen = {};
    const next = (k) => (seen[k] = (seen[k] || 0) + 1) - 1;
    sc.elements.forEach((el) => {
      if (el.type === "action") { el.addr = anchor + "..[p][" + next("p") + "]"; return; }
      if (el.type === "transition") { el.addr = anchor + "..[bq][" + next("bq") + "]"; return; }
      if (el.type === "note" || el.type === "alt") { el.addr = anchor + "..[comment][" + next("cm") + "]"; return; }
      if (el.type === "pagebreak") { el.addr = anchor + "..[hr][0]"; return; }
      if (el.type === "special") { el.addr = "..*[h3][" + (h3++) + "]"; return; }
      if (el.type === "section") { el.addr = "..*[h4][" + (h4++) + "]"; return; }
      if (el.type === "speech") {
        el.cue.addr = anchor + "._b[b.t==" + JSON.stringify(el.cue.name) + "][" + next(el.cue.name) + "]";
        el.addr = el.cue.addr;
        el.lines.forEach((ln, k) => { ln.addr = el.cue.addr + "#L" + k; });
        return;
      }
      const t = el.type.toUpperCase();
      el.addr = anchor + "._b[b.t==" + JSON.stringify(t) + "][" + next(t) + "]";
    });
  });
  return ir;
}

  const usable = (s.paperH - s.marginTop - s.marginBottom) * 72;             // pt
  const lh100 = Math.round(s.fontSize * s.lineSpacing * 100);               // 行高 ×100（整数）
  return Math.floor((usable * 100) / lh100);
}
const SPACING_MULT = { "very-tight": 0.90, tight: 0.95, normal: 1.00, loose: 1.07 };

/** 按每行字符数折行；breakAtSentences 时优先在句末断 */
const unitOf = (ch) => (ch.charCodeAt(0) > 0x2e7f ? 1.67 : 1);   // 全角/中日韩 = 12pt ÷ 7.2pt ≈ 1.67 单位（Courier 每单位 7.2pt）
const unitsOf = (s2) => { let n = 0; for (const ch of s2) n += unitOf(ch); return n; };
function wrapPlain(t, cpl) {
  const out = []; let rest = t;
  while (unitsOf(rest) > cpl) {
    let cut = -1; { let acc = 0; for (let k = 0; k < rest.length; k += 1) { const w = unitOf(rest[k]); if (acc + w > cpl) break; acc += w; if (rest[k] === " ") cut = k; } }
    if (cut < 0) { let k = 0, a = 0; for (; k < rest.length; k += 1) { const w = unitOf(rest[k]); if (a + w > cpl) break; a += w; } cut = Math.max(1, k); }
    // 避头尾：切点后若以禁则标点开头（或切点前以左括号结尾），切点前移
    while (cut > 0 && /^[，。！？、；：）】》」』…—]/.test(rest.slice(cut))) cut -= 1;
    while (cut > 0 && /[（【《「『]$/.test(rest.slice(0, cut))) cut -= 1;
    if (cut <= Math.floor(cpl * 0.5)) cut = cpl;
    out.push(rest.slice(0, cut).trimEnd()); rest = rest.slice(cut).trimStart();
  }
  if (rest) out.push(rest);
  return out;
}
/** 折行：breakAtSentences 时先按句切分再折行（句不被行/页切开 —— 手册 Break at Sentences 的语义） */
function wrap(text, cpl, breakAtSentences) {
  const t = String(text == null ? "" : text);
  if (t.length <= cpl) return [t];
  if (!breakAtSentences) return wrapPlain(t, cpl);
  const t2 = String(t).replace(/(\d)\.(\d)/g, "$1\u0001$2").replace(/\b(Mr|Mrs|Ms|Dr|St|Jr|Sr|vs|etc|Fig|No|Inc|Ltd|Co)\.|\b([egi])\.(?=[a-z])/g, "$1$2\u0001");   // 小数点与常见缩写不当句末
  const sentences = t2.match(/[^。！？.!?]+[。！？.!?]?/g) || [t2];

  const restore = (x) => x.split("\u0001").join(".");
  const out = [];
  for (const sen of sentences) {
    const ls = wrapPlain(sen.trim(), cpl);
    ls.forEach((l, k) => {
      if (k === 0 && out.length && (unitsOf(out[out.length - 1]) + 1 + unitsOf(l)) <= cpl) out[out.length - 1] += " " + l;
      else out.push(l);
    });
  }
  return out.map(restore);   // 还原被占位符保护的小数点/缩写
}

function colOf(s) { return (s.paperW - s.marginLeft - s.marginRight) * 72; }   // 版心宽度（pt）——跨版心的量都必须由它推导（KI-0c）
function cplOf(s, type) { const g = s.geo[type] || {}; const w = (g.width && type !== 'action' && type !== 'centered') ? g.width : colOf(s); return Math.max(6, Math.round(w / (0.5996 * s.fontSize))); }   // 每行预算由【该元素自己的宽度】推导：跨版心的随版心（action），绝对标准定位的固定（dialogue 252pt ⇒ 35）
function rowsOf(ir, s) {
  const rows = [];
  const push = (r) => rows.push(r);
  const emit = (el, scene, si, ei) => {
    const gap = (s.spaceBefore[el.type] != null ? s.spaceBefore[el.type] : 1);
    const prevS = (ei > 0 && scene.elements[ei - 1] && scene.elements[ei - 1].type === 'speech') ? scene.elements[ei - 1] : null;   // 成对判据
    const nextS = (scene.elements[ei + 1] && scene.elements[ei + 1].type === 'speech') ? scene.elements[ei + 1] : null;
    const sideOf = (el.cue && el.cue.dual && prevS) ? 'R' : ((el.type === 'speech' && nextS && nextS.cue.dual) ? 'L' : null);
    const base = { scene: si, sceneNo: scene ? scene.heading.num : null, gap: gap, el: el, dualSide: sideOf, lines: [""] };   // 该说话块的**每一行**都带侧标记
    if (el.type === "pagebreak") { push(Object.assign({}, base, { kind: "pagebreak", text: "", wrap: 1, geo: null })); return; }
    if (el.type === "action") {
      const g = el.subs && el.subs.indexOf("CENTER") >= 0 ? Object.assign({}, s.geo.centered, { width: colOf(s) }) : Object.assign({}, s.geo.action, { width: colOf(s) });
      const lines = wrap(el.text, cplOf(s, 'action'), s.breakAtSentences);   // 折行只用于分页计数
      push(Object.assign({}, base, { kind: "action", text: el.text, lines: lines, wrap: lines.length, geo: g, gap: gap }));
      return;
    }
    if (el.type === "speech") {
      // 成对判据（规范 §4.5）：^ 只与紧邻的前一条 cue 成对；右边那块的标记来自它自己
      const prevEl = (ei > 0 && scene.elements[ei - 1] && scene.elements[ei - 1].type === "speech") ? scene.elements[ei - 1] : null;
      const nextEl = (scene.elements[ei + 1] && scene.elements[ei + 1].type === "speech") ? scene.elements[ei + 1] : null;
      const side = (el.cue.dual && prevEl) ? "R" : ((nextEl && nextEl.cue.dual) ? "L" : null);
      const cue = { dualSide: side, kind: "character", text: el.cue.name + (el.cue.ext ? " " + el.cue.ext : ""), wrap: 1, geo: s.geo.character, gap: gap, speech: el, dual: el.cue.dual };
      push(Object.assign({}, base, cue));
      el.lines.forEach((ln, li) => {
        const g = ln.type === "parenthetical" ? s.geo.parenthetical : s.geo.dialogue;
        const cpl = ln.type === "parenthetical" ? cplOf(s, 'parenthetical') : cplOf(s, 'dialogue');
        const txt = ln.type === "parenthetical" && !/^[（(]/.test(ln.text) ? "(" + ln.text + ")" : ln.text;
        const w = wrap(txt, cpl, s.breakAtSentences);
        push(Object.assign({}, base, { kind: ln.type, text: txt, lines: w, wrap: w.length, geo: g, gap: 0, speech: el, lineIndex: li }));
      });
      return;
    }
    if (el.type === "transition") { push(Object.assign({}, base, { kind: "transition", text: el.text, wrap: 1, geo: Object.assign({}, s.geo.transition, { width: colOf(s) }), gap: gap })); return; }
    if (el.type === "lyric") {
      const g = el.subs && el.subs.indexOf("CENTER") >= 0 ? s.geo.centered : s.geo.lyric;
      const wl = wrap(el.text, cplOf(s, 'lyric'), s.breakAtSentences);
      push(Object.assign({}, base, { kind: "lyric", text: el.text, lines: wl, wrap: wl.length, geo: g, gap: gap }));
      return;
    }
    if (el.type === "section") return;   // 幕/序列仅供导航，编剧稿不进纸面（样式规范 §10）
    if (el.type === "special") { push(Object.assign({}, base, { kind: el.type, text: el.text, wrap: 1, geo: Object.assign({}, s.geo.slug, { width: colOf(s) }), gap: gap })); return; }
    if (el.type === "note" || el.type === "alt") return;   // 不进纸面
    push(Object.assign({}, base, { kind: el.type, text: el.text || "", wrap: 1, geo: s.geo.action, gap: gap }));
  };
  ir.scenes.forEach((sc, si) => {
    if (sc.heading && sc.heading.text) {
      push({ kind: "slug", text: sc.heading.text, wrap: 1, geo: Object.assign({}, s.geo.slug, { width: colOf(s) }), gap: si === 0 ? 0 : ((ir.scenes[si - 1] && ir.scenes[si - 1].elements.length === 0) ? 1 : s.spaceBefore.scene), scene: si, sceneNo: sc.heading.num, el: { type: "slug", addr: sc.heading.readAddr, num: sc.heading.num } });
    }
    sc.elements.forEach((el, ei) => emit(el, sc, si, ei));
  });
  return rows;
}

/** 分页：贪心装填 + 孤儿/寡妇规则 */
const NL = String.fromCharCode(10);
function paginate(rows, capacity, s) {
  // 预留：derive 阶段才插入的派生行（(MORE)/CONTINUED:/重复 cue）与元素间距不占 paginate 的账 ⇒ 按实测预留 6 行
  const cap = capacity - 6;
  const pages = []; let cur = []; let used = 0;
  const flush = () => { if (cur.length) { pages.push(cur); cur = []; used = 0; } };
  for (let i = 0; i < rows.length; i += 1) {
    const r = rows[i];
    if (r.kind === "pagebreak") { flush(); continue; }
    const need = (cur.length ? r.gap : 0) + r.wrap;
    const idx = cur.length;
    // 规则 1：场景头不得落在页底（其后至少 2 行同页）
    const rule = PAGINATE_AS[r.kind] || "self";
    const tailAfter = rule === "keepWithNext" ? rows.slice(i + 1, i + 3).reduce((n, x) => n + x.gap + x.wrap, 0) : 0;   // 含元素间空行
    if (START_NEW_PAGE[r.kind] || (s.startNewPage && s.startNewPage[r.kind])) { if (cur.length) flush(); }
    // 规则 2：cue 与它第一行台词同页；规则 3：副语言与所属台词同页；规则 4：台词末行不孤悬
    const prev = cur.length ? cur[cur.length - 1] : null;
    const keepWithPrev = (r.kind === "dialogue" || r.kind === "parenthetical") && prev && (prev.kind === "character" || prev.kind === "parenthetical");
    const lonelyTail = (r.kind === "character") ? 0 : 0;
    const fits = used + need + tailAfter <= cap && (!keepWithPrev || true);
    // 块可分：台词/动作/歌词块放不下时，按显示行边界切成两块（两块同属一个 speech ⇒ 页底 (MORE)、次页 CONT'D）
    if (!fits && r.lines && r.lines.length > 1) {
      const room = cap - used - (cur.length ? r.gap : 0);
      if (room < 1 && cur.length) { flush(); i -= 1; continue; }
      if (room >= 1 && room < r.lines.length) {
        const a = Object.assign({}, r, { text: r.lines.slice(0, room).join(NL), lines: r.lines.slice(0, room), wrap: room });
        const b = Object.assign({}, r, { text: r.lines.slice(room).join(NL), lines: r.lines.slice(room), wrap: r.lines.length - room, gap: 0, continued: true });
        cur.push(a); used += (cur.length > 1 ? r.gap : 0) + room;
        flush();
        rows.splice(i + 1, 0, b);
        continue;
      }
    }
    if (!fits || (keepWithPrev && used + need > cap)) {
      // 把「cue + 首行」整体推到下一页：若上一行是本页末尾的 cue，一并带走
      if (prev && prev.kind === "character" && (r.kind === "dialogue" || r.kind === "parenthetical")) { cur.pop(); used -= prev.wrap + (cur.length ? 0 : 0); pages.push(cur); cur = [prev]; used = prev.wrap; }
      else flush();
      if (cur.length === 0) used = 0;
    }
    cur.push(r); used += (cur.length > 1 ? r.gap : 0) + r.wrap;
  }
  flush();
  // 规则 4 收尾：若某页末尾只剩 cue 而无台词行，把它挪到下一页
  for (let p = 0; p < pages.length - 1; p += 1) {
    const pg = pages[p];
    if (pg.length >= 2 && pg[pg.length - 1].kind === "character" && pages[p + 1][0] && (pages[p + 1][0].kind === "dialogue" || pages[p + 1][0].kind === "parenthetical")) {
      const moved = pg.pop(); pages[p + 1].unshift(moved);
    }
  }
  return pages;
}

/** 派生：页码 · (MORE)/(CONT'D) · 整场跨页的 (CONTINUED)/CONTINUED: */
function derive(pages, s) {
  const out = [];
  for (let p = 0; p < pages.length; p += 1) {
    const pg = pages[p].slice();
    const last = pg[pg.length - 1];
    const next = pages[p + 1] ? pages[p + 1][0] : null;
    const derived = [];
    const speechSplit = last && (last.kind === "dialogue" || last.kind === "parenthetical") && next && (next.kind === "dialogue" || next.kind === "parenthetical") && last.speech === next.speech;
    if (speechSplit) {
      derived.push({ kind: "derived", text: "(MORE)", align: "right", pos: "bottom" });
      // 续页页首重复该 cue 并加 (CONT'D)（FD 惯例）—— 而不是把标记加在起始页的 cue 上
      const cue = pg.filter((r) => r.kind === "character" && r.speech === last.speech).slice(-1)[0] || pg.filter((r) => r.kind === "character").slice(-1)[0];
      if (cue) pages[p + 1] = [Object.assign({}, cue, { gap: 0, contd: true, continued: true })].concat(pages[p + 1]);
    }
    const sceneSplit = last && next && last.scene === next.scene && last.kind !== "pagebreak";
    if (s.sceneContinued && sceneSplit) {
      derived.push({ kind: "derived", text: "(CONTINUED)", pos: "bottom" });
      const no = (last.sceneNo || "").replace(/#/g, "");
      out.push(Object.assign({}, { rows: pg, derived: derived, pageNo: p + 1, topDerived: [] }));
      pages[p + 1] = [{ kind: "derived", text: "CONTINUED:" + (no ? " " + no : ""), pos: "top" }].concat(pages[p + 1]);
      continue;
    }
    out.push({ rows: pg, derived: derived, pageNo: p + 1, topDerived: [] });
  }
  // 回流校正：derive 插入的派生行（CONTINUED: / 重复 cue）此前未计入容量 ⇒ 超出的页把末尾行顺延到下一页
  for (let p2 = 0; p2 < out.length; p2 += 1) {
    const usedOf = (pg) => pg.rows.reduce((n, r, k) => n + (k === 0 ? 0 : r.gap) + r.wrap, 0);
    let guard = 0;
    while (usedOf(out[p2]) > capacityLines(s) && guard < 50) {
      guard += 1;
      const rows = out[p2].rows;
      let m = rows.length - 1;
      while (m > 0 && (rows[m].kind === "derived" || rows[m].kind === "slug")) m -= 1;
      if (m <= 0) break;
      const moved = rows.splice(m, 1)[0];
      if (p2 + 1 >= out.length) { rows.push(moved); break; }
      const nxt = out[p2 + 1].rows;
      let at = 0; while (at < nxt.length && nxt[at].kind === "derived") at += 1;
      nxt.splice(at, 0, moved);
    }
  }
  if (!s.sceneContinued) out.forEach((pg, i) => { pg.pageNo = i + 1; });
  return out;
}

/** 标题页行（样式规范 §6）：片名在 1/3 处居中 · 底部左联系方式 · 底部右稿次日期；不编号 */
function titlePageRows(ir) {
  const f = {};
  for (const x of ir.title.fields) f[x.key] = x.lines.join(' / ');
  const rows = [];
  if (ir.title.name) rows.push({ kind: 'title-name', key: '', text: String(ir.title.name).toUpperCase(), top: 264, align: 'center' });   // key='' ⇒ 写回 h1
  if (f.CREDIT) rows.push({ kind: 'title-credit', key: 'CREDIT', text: f.CREDIT, align: 'center' });
  const w = f.WRITER || f.AUTHOR || f.BY || '';
  if (w) rows.push({ kind: 'title-writer', key: 'WRITER', text: w, align: 'center' });
  const left = [f.CONTACT, f.NOTES, f.COPYRIGHT].filter(Boolean).join('  ·  ');
  if (left) rows.push({ kind: 'title-contact', text: left, pos: 'bottom-left' });
  const right = [f.DRAFT, f.DATE].filter(Boolean).join(' - ');
  if (right) rows.push({ kind: 'title-draft', key: 'DRAFT', text: right, pos: 'bottom-right' });
  return rows;
}

/** 场景表（Navigator）：每场一行，数据全部派生（零新存储） */
function navRows(doc) {
  const out = [];
  const scenes = (doc.ir && doc.ir.scenes) || [];
  let seq = 0;   // 只数真实场景（文首伪场景不算）
  scenes.forEach((sc, si) => {
    if (!sc.heading || !sc.heading.text) return;   // 文首「无场景头」的伪场景不算一场
    const mine = doc.rows.filter((r) => r.scene === si);
    const pages = [];
    for (const pg of doc.pages) if (pg.rows.some((r) => r.scene === si)) pages.push(pg.pageNo);
    const slug = String((sc.heading && sc.heading.text) || '');
    const m = /^(INT\.\/EXT\.|INT\.|EXT\.|I\/E|EST\.)\s*(.*?)\s*(?:-\s*(.*))?$/.exec(slug) || [];
    const sp = sc.elements.filter((e) => e.type === 'speech');
    seq += 1;
    out.push({ no: (sc.heading && sc.heading.num) || seq, intro: (m[1] || '').trim(), loc: (m[2] || slug).trim(), time: (m[3] || '').trim(),
      pages: pages.length ? (pages.length === 1 ? String(pages[0]) : pages[0] + '-' + pages[pages.length - 1]) : '-',
      lines: mine.length, chars: new Set(sp.map((e) => e.cue.name)).size,
      dialogues: sp.reduce((n, e) => n + e.lines.filter((l) => l.type === 'dialogue').length, 0) });
  });
  return out;
}

/** 找出成对的双人块（左块 = 前一条 speech，右块 = 带 ^ 的那条）：返回 [{left, right}]（IR 节点） */
function dualPairs(ir) {
  const out = [];
  for (const sc of ir.scenes) {
    sc.elements.forEach((el, ei) => {
      if (el.type !== 'speech' || !el.cue.dual) return;
      const prev = sc.elements[ei - 1];
      if (prev && prev.type === 'speech') out.push({ left: prev, right: el });
    });
  }
  return out;
}

function layoutDoc(ir, over) {
  const s = Object.assign({}, DEFAULT, over || {});
  s.cpl = Object.assign({}, DEFAULT.cpl, (over && over.cpl) || {});
  s.geo = Object.assign({}, DEFAULT.geo, (over && over.geo) || {});
  const cap = capacityLines(s);
  const rows = rowsOf(ir, s);
  const pages = derive(paginate(rows, cap, s), s);
  return { settings: s, capacity: cap, pages: pages, rows: rows, ir: ir, titleRows: (ir.title.name || ir.title.fields.length) ? titlePageRows(ir) : [] };
}

const quote = (s) => JSON.stringify(String(s == null ? "" : s));

/** 写前对账：盘上当前文本必须与预期一致，否则拒写 */
function reconcile(actual, expected) {
  const norm = (s) => String(s == null ? "" : s).trim();
  return { ok: norm(actual) === norm(expected), actual: norm(actual), expected: norm(expected) };
}

/** 单行节点（动作/转场/元素 bullet）⇒ 一条写回命令 */
function planLineWrite(node, newText) {

}

/** 多行说话块 ⇒ 读节体 → 拼接 → 整节写回（写回是行级的，不能直接写多行值） */
function spliceSpeech(sceneBody, speechAddr, speechMd) {
  const lines = String(sceneBody).split(String.fromCharCode(10));
  // 按地址里的出现序号定位（同一个场景里可能有两个同名 cue —— 只按名字找会改错块）
  const mi = /\[(\d+)\]/.exec(String(speechAddr));
  const wantIdx = mi ? Number(mi[1]) : 0;
  let seen = 0, start = -1;
  for (let i2 = 0; i2 < lines.length; i2 += 1) {
    const t2 = lines[i2].trim();
    const loc = speechMd.locator || speechMd.name;   // 定位用【旧】名字：改名编辑时新名在节体里还不存在
    const isCue = t2 === "- " + loc + ":" || t2 === loc || t2.startsWith("- " + loc + " ");
    if (isCue) { if (seen === wantIdx) { start = i2; break; } seen += 1; }
  }
  if (start < 0) return { ok: false, reason: "节体里找不到该说话块", body: sceneBody };
  let end = start + 1;
  while (end < lines.length && (/^\s{2,}\S/.test(lines[end]) || (lines[end].trim() === "" && end + 1 < lines.length && /^\s{2,}\S/.test(lines[end + 1])))) end += 1;   // 段内空行也算块内
  const old = lines.slice(start, end);
  const next = lines.slice(0, start).concat(speechMd.lines, lines.slice(end));
  return { ok: true, old: old, next: next.join(String.fromCharCode(10)) };
}

/** 说话块 ⇒ md 行（cue + 缩进内容） */
function speechToMd(speech) {
  const out = ["- " + speech.cue.name + (speech.cue.ext ? " " + speech.cue.ext : "") + (speech.cue.dual ? " ^" : "") + ":"];
  let prevTy = null;
  for (const ln of speech.lines) {
    if (ln.type === 'dialogue' && prevTy === 'dialogue') out.push('');   // 两段台词之间必须空行分段
    const t = ln.text;
    const isParen = (t2) => /^[（(]/.test(String(t2).trim()) && /[）)]$/.test(String(t2).trim());   // 与 ir.mjs 的 isParenLine 同义 —— 两处必须一致（有 check-dup.mjs 守着）
    const isStar = (t2) => /^\*[^*]+\*$/.test(String(t2).trim());                                  // 与 ir.mjs 的 isStarLine 同义
    const star = ln.type === 'parenthetical' && !isParen(t) && !isStar(t);
    const raw = star ? '*' + t + '*' : t;
    String(raw).split(String.fromCharCode(10)).forEach((one) => out.push('    ' + one));   // 段内每行都缩进
    prevTy = ln.type;
  }
  return { name: speech.cue.name, lines: out, md: out.join(String.fromCharCode(10)) };
}

/** 判据用：把 IR 里所有可写节点列出来 */
function writableNodes(ir) {
  const out = [];
  const walk = (el, where) => {
    if (!el) return;
    if (el.type === "action" || el.type === "transition" || el.type === "shot" || el.type === "gen") out.push({ kind: el.type, addr: el.addr, text: el.text, scene: where });
    else if (el.type === "speech") out.push({ kind: "speech", addr: el.addr, text: el.cue.name, scene: where, lines: el.lines.length });
  };
  ir.scenes.forEach((sc, si) => sc.elements.forEach((el) => walk(el, si)));
  return out;
}

/** 当前元素 + 按键 ⇒ 下一个元素 */
function nextElement(type, key) {
  if (key === 'Enter') return ENTER_FLOW[type] || 'action';
  if (key === 'Tab') return TAB_FLOW[type] || 'action';
  return null;
}

/** 反查：这个按键是哪个元素的快捷键 */
function elementOfKey(combo) {
  for (const k of Object.keys(ELEMENT_KEYS)) if (ELEMENT_KEYS[k] === combo) return k;
  return null;
}

      {view === 'paper' && doc.titleRows && doc.titleRows.length ? (
        <div className='paper-sheet paper-title-page' style={Object.assign({}, sheet, { minHeight: S.paperH + 'in' })}>
          <div style={{ position: 'absolute', top: '264pt', left: 0, right: 0, textAlign: 'center' }}>
            {doc.titleRows.filter((t) => t.align === 'center').map((t, i) => { const tk = 'T:' + (t.key || 'name');
              const tst = { display: 'block', width: 'fit-content', minWidth: '6em', minHeight: '1.5em', textTransform: t.kind === 'title-name' ? 'uppercase' : 'none', marginTop: i ? '12pt' : 0, cursor: 'text', padding: '4pt 0', margin: '0 auto' };
if (editing === tk) return <div key={tk} className={'el-' + t.kind} data-k={tk} style={tst} contentEditable suppressContentEditableWarning spellCheck={false} ref={(n) => { if (n && n.dataset.eseq !== String(EDIT_SEQ)) { n.dataset.eseq = String(EDIT_SEQ); n.replaceChildren(document.createTextNode(t.text)); n.focus(); caretFrom(n, CARET); CARET = null; } }} onInput={(e) => { const v = e.target.innerText; if (v !== LAST[tk]) { LAST[tk] = v; saveTitle(t, v, true); } }} onBlur={(e) => { const v = e.target.innerText; const ch = v !== LAST[tk]; setEditing(null); if (ch && String(v).trim()) { LAST[tk] = v; saveTitle(t, v); } }} onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); e.target.blur(); } else if (e.key === 'Escape') { e.target.innerText = t.text; LAST[tk] = t.text; saveTitle(t, t.text); e.target.blur(); } }} />;
              return <div key={tk} className={'el-' + t.kind} data-k={tk} style={tst} onMouseDown={(e) => { EDIT_SEQ += 1; LAST[tk] = t.text; CARET = { x: e.clientX, y: e.clientY }; setEditing(tk); }}>{t.text}</div>; })}
          </div>
          {doc.titleRows.filter((t) => t.pos === 'bottom-left').map((t, i) => <div key={'l' + i} className='el-title-contact' style={{ position: 'absolute', bottom: '72pt', left: '108pt' }}>{t.text}</div>)}
          {doc.titleRows.filter((t) => t.pos === 'bottom-right').map((t, i) => <div key={'r' + i} className='el-title-draft' style={{ position: 'absolute', bottom: '72pt', right: '72pt' }}>{t.text}</div>)}
        </div>
      ) : null}
      {view === 'paper' ? doc.pages.map((pg) => (
        <div className='paper-sheet' key={pg.pageNo} data-page={pg.pageNo} style={Object.assign({}, sheet, { minHeight: S.paperH + 'in' })}>
          {pg.pageNo >= 1 ? <div className='paper-page-no' style={{ position: 'absolute', top: '.55in', right: S.marginRight + 'in' }}>{pg.pageNo}.</div> : null}
          <div>
            {pageRows(pg)}
            {pg.derived.map((d, i) => <div key={'d' + i} className='el-derived' style={{ textAlign: 'right', marginTop: '12pt' }}>{d.text}</div>)}
          </div>
        </div>
      )) : null}
    </div>
  );
}
